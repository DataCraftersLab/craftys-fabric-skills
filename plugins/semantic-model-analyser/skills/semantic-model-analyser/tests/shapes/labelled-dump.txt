// Region Lookup
let
    Source = Table.FromRows({{"EU", "Europe"}, {"NA", "North America"}}, {"Code", "Name"})
in
    Source

// Region Sales
let
    Source = Orders,
    Joined = Table.NestedJoin(Source, {"Region"}, #"Region Lookup", {"Code"}, "r", JoinKind.LeftOuter),
    Grouped = Table.Group(Joined, {"Region"}, {{"Total", each List.Sum([Amount]), type number}})
in
    Grouped
